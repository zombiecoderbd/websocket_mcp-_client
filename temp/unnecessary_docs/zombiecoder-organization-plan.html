🗄️ Database Design

Proposed Schema

-- Core Tables CREATE TABLE agents ( id TEXT PRIMARY KEY, name TEXT NOT NULL, persona_type TEXT NOT NULL, description TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE conversations ( id TEXT PRIMARY KEY, agent_id TEXT REFERENCES agents(id), user_id TEXT, title TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE messages ( id TEXT PRIMARY KEY, conversation_id TEXT REFERENCES conversations(id), role TEXT NOT NULL, -- 'user', 'assistant', 'system' content TEXT NOT NULL, metadata JSON, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE models ( id TEXT PRIMARY KEY, name TEXT NOT NULL, provider TEXT NOT NULL, config JSON, is_active BOOLEAN DEFAULT TRUE, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE agent_model_mapping ( agent_id TEXT REFERENCES agents(id), model_id TEXT REFERENCES models(id), priority INTEGER DEFAULT 1, PRIMARY KEY (agent_id, model_id) ); -- Configuration Tables CREATE TABLE system_config ( key TEXT PRIMARY KEY, value JSON, description TEXT, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE user_preferences ( user_id TEXT, key TEXT, value JSON, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (user_id, key) );
